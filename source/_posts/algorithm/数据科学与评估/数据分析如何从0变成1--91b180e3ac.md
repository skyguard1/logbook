  if feature_name == "fare":
    normalizer_fn = lambda x: (x-dftrain['fare'].min())/(dftrain['fare'].max()-dftrain['fare'].min())
  else:
    normalizer_fn = None
    
  feature_columns.append(tf.feature_column.numeric_column(feature_name, dtype=tf.float32, normalizer_fn=normalizer_fn))

for feature_name, boundaries in BUCTETIZE_COLUNMS.items():
  feature_columns.append(tf.feature_column.bucketized_column(tf.feature_column.numeric_column(feature_name, dtype=tf.float32), boundaries))

def make_input_fn(data_df, label_df, num_epochs=10, shuffle=True, batch_size=32):
  def input_function():
    ds = tf.data.Dataset.from_tensor_slices((dict(data_df), label_df))
    if shuffle:
      ds = ds.shuffle(1000)
    ds = ds.batch(batch_size).repeat(num_epochs)
    return ds
  return input_function
